SELECT 
  p.subject_id,
  p.gender,
  p.dob,
  a.hadm_id,
  a.admittime,
  i.stay_id,
  i.intime AS icu_intime,
  d.icd_code,
  dd.long_title
FROM mimiciv.patients p
JOIN mimiciv.admissions a 
  ON p.subject_id = a.subject_id
JOIN mimiciv.icustays i 
  ON a.hadm_id = i.hadm_id
JOIN mimiciv.diagnoses_icd d 
  ON a.hadm_id = d.hadm_id
JOIN mimiciv.d_icd_diagnoses dd 
  ON d.icd_code = dd.icd_code AND d.icd_version = dd.icd_version
WHERE 
  a.admittime BETWEEN '2021-10-01' AND '2022-10-31'
  AND LOWER(dd.long_title) SIMILAR TO '%(pregnan|labor|delivery|obstetric|fetal)%'
  AND DATE_PART('year', a.admittime) - DATE_PART('year', p.dob) BETWEEN 12 AND 55

select did.long_title, count(1)
from   admissions a 
join   patients p
on     a.subject_id = p.subject_id
join   diagnoses_icd d
on     a.subject_id  = d.subject_id
and    a.hadm_id     = d.hadm_id
join   d_icd_diagnoses did 
on     d.icd_code    = did.icd_code
where  a.deathtime is not null
and    p.gender = 'F'
and    p.dob is not null
group by did.long_title
order by 2 desc

select sum(case when DATE_PART('day', X.minimo - X.dob) >= -3 then 1 else 0 end) as suma_minimo_3dias
      , sum(case when DATE_PART('day', X.maximo - X.dob) >= -3 then 1 else 0 end) as suma_maximo_3dias
      , sum(case when DATE_PART('day', X.minimo - X.dob) >= -7 then 1 else 0 end) as suma_minimo_7dias
      , sum(case when DATE_PART('day', X.maximo - X.dob) >= -7 then 1 else 0 end) as suma_maximo_7dias
      , sum(case when DATE_PART('day', X.minimo - X.dob) >= -15 then 1 else 0 end) as suma_minimo_15dias
      , sum(case when DATE_PART('day', X.maximo - X.dob) >= -15 then 1 else 0 end) as suma_maximo_15dias
      , sum(case when DATE_PART('day', X.maximo - X.dob) < 0 then 1 else 0 end) as UCI_Antes_Nacimiento
      
      , count(1)
from 
(
select  p.subject_id, p.dob, min(i.intime) as minimo, max(i.intime) as maximo
from    patients p
join    icustays i 
on      i.subject_id = p.subject_id 
where   p.dob  is not null
and    TO_CHAR(p.dob, 'YYYYMM')  >= '211001' 
and    TO_CHAR(p.dob, 'YYYYMM')  <= '221410'
group by p.subject_id, p.dob
) X
